        for model_name in MODEL_CHAIN:
            for key_number, api_key in available_keys:
                start_time = time.time()

                        api_key = [REDACTED]"text", None)

                    if not text or not text.strip():

                    logger.info(

                    )

                    return {
                        "success": True,
                        "text": text.strip(),
                        "model": model_name,
                        "key_slot": key_number,

                        type(error).__name__,
                        elapsed,
                    )

                    errors.append({
                        "model": model_name,
                        "key": key_number,
                        "error": type(error).__name__,
                    })

        raise AIServiceError(
            "Seluruh model Gemini gagal memberikan respons."

def analyze_education(year=None):
    df = load_dataset()

    available_years = sorted(
        df["tahun"].dropna().astype(int).unique().tolist()
    )

    if not available_years:
        return {
            "error": "Dataset tidak memiliki data tahun.",
        }

    latest_year = max(available_years)

    if year is None:
        year = latest_year

    year = int(year)

    current = df[df["tahun"] == year].copy()

    if current.empty:
        return {
            "error": f"Data tahun {year} tidak tersedia.",
            "available_years": available_years,
        }

    previous_year = year - 1
    previous = df[df["tahun"] == previous_year].copy()

    indicator_summary = []
    level_summary = []
    district_summary = []
    change_summary = []

    for column, config in INDICATORS.items():
        values = current[column].dropna()

        if values.empty:
            continue

        highest_idx = current[column].idxmax()
        lowest_idx = current[column].idxmin()

        highest_row = current.loc[highest_idx]
        lowest_row = current.loc[lowest_idx]

        indicator_summary.append({
            "column": column,
            "label": config["label"],
            "category": config["category"],
            "level": config["level"],
            "mean": float(values.mean()),
            "median": float(values.median()),
            "min": float(values.min()),
            "max": float(values.max()),
            "highest_district": highest_row["bps_nama_kecamatan"],
            "highest_value": float(highest_row[column]),
            "lowest_district": lowest_row["bps_nama_kecamatan"],
            "lowest_value": float(lowest_row[column]),
            "data_count": int(values.count()),
        })

    for level in ["SMA", "MA", "SMK"]:
        level_indicators = [
            item for item in indicator_summary
            if item["level"] == level
        ]

        level_summary.append({
            "level": level,
            "indicator_count": len(level_indicators),
            "indicators": level_indicators,
        })

    for _, row in current.iterrows():
        district_summary.append({
            "district": row["bps_nama_kecamatan"],
            "year": int(row["tahun"]),
            "indicators": {
                column: float(row[column])
                for column in INDICATORS
                if column in row and row[column] is not None
            },
        })

    if not previous.empty:
        previous = previous.set_index("bps_nama_kecamatan")

        for column, config in INDICATORS.items():
            for _, row in current.iterrows():
                district = row["bps_nama_kecamatan"]

                if district not in previous.index:
                    continue

                old_value = float(previous.loc[district, column])
                new_value = float(row[column])
                difference = new_value - old_value

                change_summary.append({
                    "district": district,
                    "column": column,
                    "label": config["label"],
                    "category": config["category"],
                    "level": config["level"],
                    "previous_year": previous_year,
                    "current_year": year,
                    "previous_value": old_value,
                    "current_value": new_value,
                    "change": difference,
                    "direction": (
                        "naik" if difference > 0
                        else "turun" if difference < 0
                        else "tetap"
                    ),
                })

    increases = sorted(
        [item for item in change_summary if item["change"] > 0],
        key=lambda item: item["change"],
        reverse=True,
    )

    decreases = sorted(
        [item for item in change_summary if item["change"] < 0],
        key=lambda item: item["change"],
    )

    return {
        "year": year,
        "latest_year": latest_year,
        "available_years": available_years,
        "total_data": len(current),
        "total_districts": current["bps_nama_kecamatan"].nunique(),
        "indicator_summary": indicator_summary,
        "level_summary": level_summary,
        "district_summary": district_summary,
        "change_summary": change_summary,
        "top_increases": increases[:10],
        "top_decreases": decreases[:10],
    }


def get_education_analysis(year=None):
    return analyze_education(year)

from pathlib import Path

import pandas as pd

DATASET_PATH = Path(__file__).resolve().parent.parent / "data" / "dataset_pendidikan.csv"

NUMERIC_COLUMNS = [
    "angka_mengulang_sma",
    "angka_putus_sekolah_sma",
    "angka_lulusan_sma",
    "angka_mengulang_ma",
    "angka_putus_sekolah_ma",
    "angka_lulusan_ma",
    "angka_mengulang_smk",
    "angka_putus_sekolah_smk",
    "angka_lulusan_smk",
]


def load_dataset():
    df = pd.read_csv(DATASET_PATH, dtype=str)

    for column in NUMERIC_COLUMNS:
        df[column] = pd.to_numeric(
            df[column].str.replace(",", ".", regex=False),
            errors="coerce"
        )

    df["tahun"] = pd.to_numeric(df["tahun"], errors="coerce")

    return df


def get_chart_data():
    df = load_dataset()

    indicators = {
        "mengulang": [
            "angka_mengulang_sma",
            "angka_mengulang_ma",
            "angka_mengulang_smk",
        ],
        "putus_sekolah": [
            "angka_putus_sekolah_sma",
            "angka_putus_sekolah_ma",
            "angka_putus_sekolah_smk",
        ],
        "lulusan": [
            "angka_lulusan_sma",
            "angka_lulusan_ma",
            "angka_lulusan_smk",
        ],
    }

    result = {}

    for category, columns in indicators.items():
        grouped = df.groupby("tahun")[columns].mean().round(2)

        result[category] = {
            "labels": [int(year) for year in grouped.index],
            "datasets": [
                {
                    "label": level,
                    "data": grouped[column].fillna(0).tolist(),
                }
                for level, column in zip(["SMA", "MA", "SMK"], columns)
            ],
        }

    return result

def parse_number(value, decimal=False):
    if pd.isna(value):
        return 0

    value = str(value).strip()
    if not value:
        return 0

    if decimal:
        value = value.replace(",", ".")
        try:
            return float(value)
        except ValueError:
            return 0

    value = value.replace(".", "").replace(",", "")
    try:
        return int(float(value))
    except ValueError:
        return 0


def load_new_dataset():
    records = []

    for school_year, filepath in FILES.items():
        if not filepath.exists():
            continue

        raw = pd.read_csv(filepath, header=None, skiprows=2, dtype=str)

        for _, row in raw.iterrows():
            if len(row) < 10:
                continue

            province = str(row.iloc[2]).strip()
            level = str(row.iloc[1]).strip()

            if not province or province.lower() == "nan":
                continue

            records.append({
                "tahun_ajar": school_year,
                "jenjang": level,
                "provinsi": province,
                "peserta_didik_baru": parse_number(row.iloc[3]),
                "jumlah_peserta_didik": parse_number(row.iloc[4]),
                "lulusan": parse_number(row.iloc[5]),
                "mengulang_jumlah": parse_number(row.iloc[6]),
                "mengulang_persentase": parse_number(row.iloc[7], decimal=True),
                "putus_jumlah": parse_number(row.iloc[8]),
                "putus_persentase": parse_number(row.iloc[9], decimal=True),
            })

    return pd.DataFrame(records)

def _calculate_stability(values):
    values = [float(v) for v in values if pd.notna(v)]

    if len(values) <= 1:
        return 50.0

    mean_value = np.mean(values)
    std_value = np.std(values)

    scale = abs(mean_value)

    if scale < 1e-6:
        scale = max([abs(v) for v in values] or [0])

    if scale < 1e-6:
        return 100.0

    score = 100 / (1 + (std_value / scale * 10))

    return round(max(0, min(100, score)), 2)


def _get_trend(values):
    if len(values) < 2:
        return {
            "direction": "Belum cukup data",
            "change": 0.0,
        }

    change = float(values[-1]) - float(values[-2])

    if change > 0:
        direction = "Naik"
    elif change < 0:
        direction = "Turun"
    else:
        direction = "Stabil"

    return {
        "direction": direction,
        "change": change,
    }


def _bounded_value(value, config):

def _regional_statistics(df, column, year=None):
    if year is None:
        year = int(df["tahun"].max())

    values = df[df["tahun"] == year][column].dropna()

    if values.empty:
        return 0.0, 0.0, 0.0

    mean_value = float(values.mean())
    median_value = float(values.median())

    positive = values[values > 0]

    positive_mean = (
        float(positive.mean()) if not positive.empty else 0.0
    )

    return mean_value, median_value, positive_mean

def calculate_indicator_forecast(
    district_data,
    full_dataset,
    column,
    config,
    target_year,
):
    historical = (
        district_data[["tahun", column]]
        .dropna()
        .sort_values("tahun")
    )

    years = historical["tahun"].astype(int).tolist()
    values = historical[column].astype(float).tolist()

    if not values:
        return 0.0, "Tidak ada data"

    current_year = years[-1]
    current_value = values[-1]

    if len(values) >= 2:
        year_gap = max(1, years[-1] - years[-2])
        annual_change = (values[-1] - values[-2]) / year_gap
    else:
        annual_change = 0.0

    related_indicators = {
        "angka_lulusan_sma": (
            "angka_mengulang_sma",
            "angka_putus_sekolah_sma",
        ),
        "angka_lulusan_ma": (
            "angka_mengulang_ma",
            "angka_putus_sekolah_ma",
        ),
        "angka_lulusan_smk": (
            "angka_mengulang_smk",
            "angka_putus_sekolah_smk",
        ),
    }

    is_graduate = config["category"] == "Lulusan"
    related_projection = {}

    if is_graduate and column in related_indicators:
        for related_column in related_indicators[column]:
            history = (
                district_data[["tahun", related_column]]
                .dropna()
                .sort_values("tahun")
            )

            related_values = history[related_column].astype(float).tolist()
            related_years = history["tahun"].astype(int).tolist()

            if not related_values:
                related_projection[related_column] = {
                    "value": 0.0,
                    "trend": 0.0,
                }
                continue

            related_trend = 0.0

            if len(related_values) >= 2:
                gap = max(1, related_years[-1] - related_years[-2])
                related_trend = (
                    related_values[-1] - related_values[-2]
                ) / gap

            related_projection[related_column] = {
                "value": related_values[-1],
                "trend": related_trend,
            }

    predicted = current_value

    # Asumsi pertumbuhan siswa moderat, bukan data aktual.
    student_growth_rate = 0.005

    for year in range(current_year + 1, int(target_year) + 1):
        yearly_change = annual_change

        if is_graduate:
            yearly_change += max(0.0, predicted) * student_growth_rate

            interaction_change = 0.0

            for projection in related_projection.values():
                previous_value = projection["value"]
                next_value = _bounded_value(
                    previous_value + projection["trend"],
                    {"category": "Rate"},
                )

                interaction_change -= 0.25 * (
                    next_value - previous_value
                )

                projection["value"] = next_value

            yearly_change = yearly_change + interaction_change

        predicted += yearly_change
        predicted = _bounded_value(predicted, config)

    method = "Local Trend + Student Growth Assumption + Dynamic Indicator Interaction"

    return predicted, method

def _build_explanation(
    config,
    current_value,
    predicted_value,
    trend,
    method_name,
    history_count,
    stability_score,
):
    change = predicted_value - current_value

    if change > 0:
        direction = "meningkat"
    elif change < 0:
        direction = "menurun"
    else:
        direction = "tetap"

    if config["category"] in ["Mengulang", "Putus Sekolah"]:
        if change > 0:
            interpretation = (
                "Nilai diproyeksikan meningkat dan perlu menjadi perhatian."
            )
        elif change < 0:
            interpretation = (
                "Nilai diproyeksikan menurun. Perubahan perlu dipantau."
            )
        else:
            interpretation = "Nilai diproyeksikan relatif tetap."
    else:
        if change > 0:
            interpretation = "Angka lulusan diproyeksikan meningkat."
        elif change < 0:
            interpretation = "Angka lulusan diproyeksikan menurun."
        else:
            interpretation = "Angka lulusan diproyeksikan tetap."

    explanation = (
        f"{config['label']} diperkirakan {direction}. "
        f"Nilai aktual terakhir {current_value:.4f}, "
        f"proyeksi {predicted_value:.4f}. "
        f"Metode: {method_name}."
    )

    method_explanation = (
        "Mesin membaca data aktual kecamatan, menghitung perubahan "
        "antar-tahun, lalu menggunakan perubahan tersebut sebagai "
        "dasar proyeksi tahun berikutnya. Prediksi dilakukan bertahap. "
        "Indikator lulusan tidak dibatasi maksimum 100."
    )

    return {
        "prediction_direction": direction,
        "interpretation": interpretation,
        "explanation": explanation,
        "method_explanation": method_explanation,
        "historical_trend": trend["direction"],
        "historical_change": trend["change"],
        "history_count": history_count,
        "stability_score": stability_score,
    }


def predict_education(district, prediction_year):
    df = load_dataset()
    prediction_year = int(prediction_year)

    if prediction_year not in ALLOWED_YEARS:
        raise ValueError(
            "Tahun prediksi hanya tersedia untuk 2026, 2027, dan 2028."
        )

    district_data = (
        df[df["bps_nama_kecamatan"] == district]
        .sort_values("tahun")
    )

    if district_data.empty:
        raise ValueError("Kecamatan tidak ditemukan dalam dataset.")

    latest_data = district_data.iloc[-1]
    actual_latest_year = int(latest_data["tahun"])

    predictions = []

    for column, config in INDICATORS.items():
        historical_data = (
            district_data[["tahun", column]]
            .dropna()
            .sort_values("tahun")
        )

        historical_values = (
            historical_data[column].astype(float).tolist()
        )

        historical_years = (
            historical_data["tahun"].astype(int).tolist()
        )

        current_value = float(historical_values[-1])

        trend = _get_trend(historical_values)
        stability_score = _calculate_stability(historical_values)

        predicted_value, method_name = calculate_indicator_forecast(
            district_data=district_data,
            full_dataset=df,
            column=column,
            config=config,
            target_year=prediction_year,
        )

        explanation_data = _build_explanation(
            config=config,
            current_value=current_value,
            predicted_value=predicted_value,
            trend=trend,
            method_name=method_name,
            history_count=len(historical_values),
            stability_score=stability_score,
        )

        predictions.append({
            "column": column,
            "label": config["label"],
            "category": config["category"],
            "level": config["level"],
            "current_value": current_value,
            "predicted_value": predicted_value,
            "change": predicted_value - current_value,
            "method": method_name,
            **explanation_data,
        })

    # Data aktual untuk grafik.
    actual_data = []

    for year in sorted(district_data["tahun"].unique()):
        year_row = district_data[district_data["tahun"] == year]

        if year_row.empty:
            continue

        row = year_row.iloc[-1]

        actual_data.append({
            "year": int(year),
            "type": "actual",
            "values": {
                column: (
                    float(row[column])
                    if pd.notna(row[column])
                    else 0.0
                )
                for column in INDICATORS
            },
        })

    # Prediksi 2026–2028.
    forecast_data = []

    for year in FORECAST_YEARS:
        year_predictions = {}

        for column, config in INDICATORS.items():
            predicted, _ = calculate_indicator_forecast(
                district_data=district_data,
                full_dataset=df,
                column=column,
                config=config,
                target_year=year,
            )

            year_predictions[column] = predicted

        forecast_data.append({
            "year": year,
            "type": "forecast",
            "values": year_predictions,
        })

    return {
        "district": district,
        "latest_year": actual_latest_year,
        "prediction_year": prediction_year,
        "actual_data": actual_data,
        "forecast_data": forecast_data,
        "predictions": predictions,
    }

def prediction_page(request):
    from dashboard.predictor import (
        get_prediction_options,
        predict_education,
        INDICATORS,
    )

    options = get_prediction_options()

    district = request.POST.get(
        "kecamatan",
        request.GET.get(
            "kecamatan",
            options["districts"][0] if options["districts"] else ""
        )
    )

    year_input = request.POST.get(
        "tahun",
        request.GET.get(
            "tahun",
            str(options["default_prediction_year"])
        )
    )

    result = None
    error = None
    chart_data = {}

    try:
        prediction_year = int(year_input)

        result = predict_education(
            district=district,
            prediction_year=prediction_year
        )

        df = load_dataset()

        district_data = df[
            df["bps_nama_kecamatan"] == district
        ].sort_values("tahun")

        actual_years = sorted(
            district_data["tahun"].dropna().astype(int).unique().tolist()
        )

        forecast_years = [2026, 2027, 2028]
        chart_years = actual_years + forecast_years

        chart_data = {
            "labels": chart_years,
            "mengulang": [],
            "putus_sekolah": [],
            "lulusan": [],
        }

        category_keys = {
            "Mengulang": "mengulang",
            "Putus Sekolah": "putus_sekolah",
            "Lulusan": "lulusan",
        }

        colors = {
            "SMA": "#2563eb",
            "MA": "#0d9488",
            "SMK": "#f59e0b",
        }

        for column, config in INDICATORS.items():
            category = category_keys[config["category"]]

            actual_values = []

            for year in actual_years:
                year_rows = district_data[
                    district_data["tahun"] == year
                ]

                value = (
                    float(year_rows.iloc[-1][column])
                    if not year_rows.empty
                    else None
                )

                actual_values.append(value)

            forecast_values = []

            for year in forecast_years:
                forecast_result = predict_education(
                    district=district,
                    prediction_year=year
                )

                prediction_item = next(
                    item for item in forecast_result["predictions"]
                    if item["column"] == column
                )

                forecast_values.append(
                    prediction_item["predicted_value"]
                )

            chart_data[category].append({
                "label": config["level"],
                "color": colors[config["level"]],
                "actual": actual_values + [None] * len(forecast_years),
                "forecast": (
                    [None] * len(actual_years)
                    + forecast_values
                ),
            })

    except (ValueError, TypeError, KeyError, IndexError) as exc:
        error = str(exc)

    context = {
        "active_page": "prediction",
        "districts": options["districts"],
        "selected_district": district,
        "latest_year": options["latest_year"],
        "prediction_year": year_input,
        "result": result,
        "chart_data": chart_data,
        "error": error,
    }

    return render(request, "dashboard/prediction.html", context)

def dashboard_ai_explanation(request):
    import hashlib
    import json

    from django.core.cache import cache
    from django.http import JsonResponse

    from dashboard.data_processor import load_dataset, get_chart_data
    from dashboard.ai_interpreter import ai_interpreter
    from dashboard.ai_service import AIServiceError

    df = load_dataset()

    statistics = {
        "total_data": len(df),
        "total_kecamatan": df["bps_nama_kecamatan"].nunique(),
        "tahun_awal": int(df["tahun"].min()),
        "tahun_akhir": int(df["tahun"].max()),
        "total_jenjang": 3,
    }

    chart_data = get_chart_data()

    cache_context = {
        "statistics": statistics,
        "charts": chart_data,
    }

    fingerprint = hashlib.sha256(
        json.dumps(
            cache_context,
            sort_keys=True,
            ensure_ascii=False
        ).encode("utf-8")
    ).hexdigest()

    cache_key = f"dashboard_ai_{fingerprint}"

    cached_result = cache.get(cache_key)

    if cached_result:
        return JsonResponse({
            **cached_result,
            "cached": True,
        })

    try:
        result = ai_interpreter.explain_dashboard(
            chart_data=chart_data,
            statistics=statistics,
        )

        response_data = {
            "success": True,
            "explanations": result["explanations"],
            "model": result["model"],
            "elapsed": result["elapsed"],
            "cached": False,
        }

        cache.set(cache_key, response_data, timeout=86400)

        return JsonResponse(response_data)

    except AIServiceError:
        return JsonResponse({
            "success": False,
            "error": "AI sedang tidak dapat memberikan analisis. Silakan coba kembali.",
        }, status=503)

    except Exception:
        return JsonResponse({
            "success": False,
            "error": "Terjadi kesalahan saat memproses analisis AI.",
        }, status=500)


def prediction_ai_explanation(request):
    import hashlib
    import json

    from django.core.cache import cache
    from django.http import JsonResponse

    from dashboard.predictor import predict_education
    from dashboard.ai_interpreter import ai_interpreter
    from dashboard.ai_service import AIServiceError

    district = request.GET.get("kecamatan", "").strip()
    year_input = request.GET.get("tahun", "").strip()

    if not district or not year_input:
        return JsonResponse({
            "success": False,
            "error": "Kecamatan dan tahun prediksi harus diisi.",
        }, status=400)

    try:
        prediction_year = int(year_input)
        prediction_data = predict_education(
            district=district,
            prediction_year=prediction_year,
        )

        cache_context = {
            "district": district,
            "year": prediction_year,
            "prediction": prediction_data,
        }

        fingerprint = hashlib.sha256(
            json.dumps(
                cache_context,
                sort_keys=True,
                ensure_ascii=False,
            ).encode("utf-8")
        ).hexdigest()

        cache_key = f"prediction_ai_{fingerprint}"

        cached_result = cache.get(cache_key)

        if cached_result:
            return JsonResponse({
                **cached_result,
                "cached": True,
            })

        result = ai_interpreter.explain_prediction(
            district=district,
            prediction_year=prediction_year,
            chart_data=prediction_data,
        )

        response_data = {
            "success": True,
            "district": district,
            "prediction_year": prediction_year,
            "explanations": result["explanations"],
            "model": result["model"],
            "elapsed": result["elapsed"],
            "cached": False,
        }

        cache.set(cache_key, response_data, timeout=86400)

        return JsonResponse(response_data)

    except AIServiceError:
        return JsonResponse({
            "success": False,
        "error": "AI sedang tidak dapat memberikan analisis. Silakan coba kembali.",
        }, status=503)

    except (ValueError, TypeError):
        return JsonResponse({
            "success": False,
            "error": "Kecamatan atau tahun prediksi tidak valid.",
        }, status=400)

    except Exception:
        return JsonResponse({
            "success": False,
            "error": "Terjadi kesalahan saat memproses analisis AI prediksi.",
        }, status=500)

function createChart(elementId, data, title) {
    const element = document.getElementById(elementId);

    new Chart(element, {
        type: "bar",
        data: {
            labels: data.labels,
            datasets: data.datasets.map((dataset) => ({
                label: dataset.label,
                data: dataset.data,
                backgroundColor: {
                    SMA: "#2563eb",
                    MA: "#0d9488",
                    SMK: "#f59e0b"
                }[dataset.label],
                borderRadius: 5,
                maxBarThickness: 45
            }))
        },
        options: {
            responsive: true,
            maintainAspectRatio: false,
            plugins: {
                legend: {
                    position: "bottom"
                },
                title: {
                    display: false,
                    text: title
                },
                tooltip: {
                    callbacks: {
                        label: function(context) {
                            return context.dataset.label + ": " +
                                context.parsed.y.toLocaleString("id-ID", {
                                    maximumFractionDigits: 2
                                }) + "%";
                        }
                    }
                }
            },
            scales: {
                y: {
                    beginAtZero: true,
                    ticks: {
                        callback: function(value) {
                            return Number(value).toLocaleString("id-ID", {
                                maximumFractionDigits: 2
                            }) + "%";
                        }
                    },
                    title: {
                        display: true,
                        text: "Persentase (%)"
                    }
                },
                x: {
                    title: {
                        display: true,
                        text: "Tahun"
                    }
                }
            }
        }
    });
}

(() => {
    const reducedMotion = window.matchMedia(
        '(prefers-reduced-motion: reduce)'
    ).matches;

    if (reducedMotion) return;

    const overlay = document.createElement('div');
    overlay.className = 'page-transition-overlay transition-reveal';
    document.body.appendChild(overlay);

    // Reveal the newly loaded page.
    window.addEventListener('pageshow', () => {
        overlay.classList.remove('transition-cover');
        overlay.classList.add('transition-reveal');
    });

    // Smooth transition for internal page navigation.
    document.addEventListener('click', (event) => {
        const link = event.target.closest('a[href]');
        if (!link) return;

        if (
            event.defaultPrevented ||
            event.button !== 0 ||
            event.ctrlKey ||
            event.shiftKey ||
            event.altKey ||
            event.metaKey ||
            link.target === '_blank' ||
            link.hasAttribute('download')
        ) return;

        const url = new URL(link.href, window.location.href);

        if (
            url.origin !== window.location.origin ||
            url.pathname === window.location.pathname &&
            url.search === window.location.search ||
            link.getAttribute('href').startsWith('#')
        ) return;

        event.preventDefault();

        overlay.classList.remove('transition-reveal');
        overlay.classList.add('transition-cover');

        setTimeout(() => {
            window.location.href = url.href;
        }, 620);
    });
})();

(function () {
    const payloadElement = document.getElementById("dashboard-chart-data");
    if (!payloadElement || typeof Chart === "undefined") return;

    let chartData;
    try {
        chartData = JSON.parse(payloadElement.textContent);
    } catch (error) {
        console.error("Dashboard chart data gagal dibaca.", error);
        return;
    }

    const palette = [
        { border: "#2864e8", background: "rgba(40,100,232,.08)" },
        { border: "#25a47a", background: "rgba(37,164,122,.08)" },
        { border: "#8a65d6", background: "rgba(138,101,214,.08)" }
    ];

    function createDashboardChart(canvasId, category) {
        const canvas = document.getElementById(canvasId);
        const source = chartData[category];

        if (!canvas || !source) return;

        const datasets = (source.datasets || []).map((dataset, index) => ({
            label: dataset.label,
            data: dataset.data,
            borderColor: palette[index].border,
            backgroundColor: palette[index].background,
            borderWidth: 2.5,
            pointRadius: 3,
            pointHoverRadius: 5,
            pointBackgroundColor: "#fff",
            pointBorderWidth: 2,
            tension: 0.36,
            fill: false
        }));

        new Chart(canvas, {
            type: "line",
            data: {
                labels: source.labels,
                datasets: datasets
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                interaction: {
                    intersect: false,
                    mode: "index"
                },
                animation: {
                    duration: 850,
                    easing: "easeOutQuart"
                },
                plugins: {
                    legend: {
                        position: "bottom",
                        labels: {
                            usePointStyle: true,
                            pointStyle: "circle",
                            boxWidth: 7,
                            boxHeight: 7,
                            padding: 16,
                            color: "#73829a",
                            font: {
                                size: 10,
                                family: "inherit"
                            }
                        }
                    },
                    tooltip: {
                        backgroundColor: "#192b57",
                        titleColor: "#fff",
                        bodyColor: "#dce7fa",
                        padding: 12,
                        cornerRadius: 9,
                        displayColors: true
                    }
                },
                scales: {
                    x: {
                        grid: { display: false },
                        border: { display: false },
                        ticks: {
                            color: "#92a0b5",
                            font: { size: 10 }
                        }
                    },
                    y: {

                        border: { display: false, dash: [4, 4] },

                        },
                        ticks: {
                            color: "#92a0b5",
                            padding: 8,
                            font: { size: 10 }
                        }
                    }
                }
            }
        });
    }

document.addEventListener("DOMContentLoaded", function () {
    const dataElement = document.getElementById("prediction-chart-data");
    if (!dataElement) return;

    const data = JSON.parse(dataElement.textContent);
    const labels = data.labels || [];
    const selectedYear = Number("{{ result.prediction_year }}");

    const colors = {
        SMA: "#2563eb",
        MA: "#0d9488",
        SMK: "#f59e0b"
    };

    function formatPercent(value) {
        if (value === null || value === undefined || !Number.isFinite(value)) {
            return "—";
        }

        return new Intl.NumberFormat("id-ID", {
            minimumFractionDigits: 0,
            maximumFractionDigits: 2
        }).format(value) + "%";
    }

    function getPeriodValues(seriesList, yearLimit) {
        const values = [];

        seriesList.forEach(series => {
            series.actual.forEach((value, index) => {
                const year = Number(labels[index]);

                if (
                    value !== null &&
                    value !== undefined &&
                    Number.isFinite(value) &&
                    year <= yearLimit
                ) {
                    values.push(value);
                }
            });

            series.forecast.forEach((value, index) => {
                const year = Number(labels[index]);

                if (
                    value !== null &&
                    value !== undefined &&
                    Number.isFinite(value) &&
                    year <= yearLimit
                ) {
                    values.push(value);
                }
            });
        });

        return values;
    }

    function calculateAverage(seriesList, yearLimit) {
        const values = getPeriodValues(seriesList, yearLimit);

        if (!values.length) return null;

        return values.reduce((sum, value) => sum + value, 0) / values.length;
    }

    function calculateTrend(seriesList, yearLimit) {
        const startValues = [];
        const endValues = [];

        seriesList.forEach(series => {
            labels.forEach((year, index) => {
                const numericYear = Number(year);

                if (numericYear === 2024) {
                    if (series.actual[index] !== null) {
                        startValues.push(series.actual[index]);
                    }
                }

                if (numericYear === yearLimit) {
                    const value =
                        series.forecast[index] !== null
                            ? series.forecast[index]
                            : series.actual[index];

                    if (value !== null && value !== undefined) {
                        endValues.push(value);
                    }
                }
            });
        });

        if (!startValues.length || !endValues.length) {
            return {
                label: "Belum dapat ditentukan",
                description: "Data awal atau data akhir periode belum tersedia."
            };
        }

        const startAverage =
            startValues.reduce((a, b) => a + b, 0) / startValues.length;

        const endAverage =
            endValues.reduce((a, b) => a + b, 0) / endValues.length;

        const difference = endAverage - startAverage;
        const tolerance = 0.01;

        if (difference > tolerance) {
            return {
                label: "Tren Meningkat",
                description:
                "Rata-rata indikator pada akhir periode lebih tinggi dibandingkan tahun 2024."
            };
        }

        if (difference < -tolerance) {
            return {
                label: "Tren Menurun",
            description:
                "Rata-rata indikator pada akhir periode lebih rendah dibandingkan tahun 2024."
            };
        }

        return {
            label: "Relatif Stabil",
            description:
            "Perubahan rata-rata dari tahun 2024 hingga akhir periode relatif kecil."
        };
    }

    function renderSummary(seriesList, averageId, trendId, descriptionId) {
        const average = calculateAverage(seriesList, selectedYear);
        const trend = calculateTrend(seriesList, selectedYear);

        document.getElementById(averageId).textContent =
            formatPercent(average);

        document.getElementById(trendId).textContent = trend.label;
        document.getElementById(descriptionId).textContent =
            trend.description;
    }

    function createLineChart(canvasId, seriesList) {
        const canvas = document.getElementById(canvasId);
        if (!canvas) return;

        const datasets = [];

        seriesList.forEach(series => {
            datasets.push({
                label: series.label + " Aktual",
                data: series.actual,
                borderColor: series.color,
                backgroundColor: series.color,
                borderWidth: 2,
                pointRadius: 4,
                pointHoverRadius: 7,
                pointStyle: "circle",
                tension: 0.25,
                spanGaps: false
            });

            datasets.push({
                label: series.label + " Prediksi",
                data: series.forecast,
                borderColor: series.color,
                backgroundColor: series.color,
                borderWidth: 2,
                borderDash: [6, 5],
                pointRadius: 5,
                pointHoverRadius: 8,
                pointStyle: "triangle",
                tension: 0.25,
                spanGaps: false
            });
        });

        new Chart(canvas, {
            type: "line",
            data: {
                labels: labels,
                datasets: datasets
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                interaction: {
                    mode: "index",
                    intersect: false
                },
                plugins: {
                    legend: {
                        position: "bottom"
                    },
                    tooltip: {
                        callbacks: {
                            label: function (context) {
                                return context.dataset.label + ": " +
                                    formatPercent(context.parsed.y);
                            }
                        }
                    }
                },
                scales: {
                    y: {
                        beginAtZero: true,
                        ticks: {
                            callback: function (value) {
                                return value + "%";
                            }
                        }
                    },
                    x: {
                        title: {
                            display: true,
                            text: "Tahun"
                        }
                    }
                }
            }
        });
    }

    function calculateYearlyAverage(seriesList, year) {
        const values = [];

        seriesList.forEach(series => {
            labels.forEach((label, index) => {
                if (Number(label) !== year) return;

                const value =
                    series.forecast[index] !== null
                        ? series.forecast[index]
                        : series.actual[index];

                if (value !== null && value !== undefined && Number.isFinite(value)) {
                    values.push(value);
                }
            });
        });

        if (!values.length) return null;

        return values.reduce((sum, value) => sum + value, 0) / values.length;
    }

    const indicatorSeries = [
        {
            label: "Mengulang",
            series: data.mengulang,
            color: "#2563eb"
        },
        {
            label: "Putus Sekolah",
            series: data.putus_sekolah,
            color: "#dc2626"
        },
        {
            label: "Lulusan",
            series: data.lulusan,
            color: "#0d9488"
        }
    ];

    createLineChart("chartMengulang", data.mengulang);
    createLineChart("chartPutus", data.putus_sekolah);
    createLineChart("chartLulusan", data.lulusan);

    renderSummary(
        data.mengulang,
        "averageMengulang",
        "trendMengulang",
        "trendDescriptionMengulang"
    );

    renderSummary(
        data.putus_sekolah,
        "averagePutus",
        "trendPutus",
        "trendDescriptionPutus"
    );

    renderSummary(
        data.lulusan,
        "averageLulusan",
        "trendLulusan",
        "trendDescriptionLulusan"
    );

    const comparisonCanvas = document.getElementById("chartComparison");

    if (comparisonCanvas) {
        const comparisonDatasets = indicatorSeries.map(indicator => ({
            label: indicator.label,
            data: labels.map(year =>
                calculateYearlyAverage(indicator.series, Number(year))
            ),
            borderColor: indicator.color,
            backgroundColor: indicator.color,
            borderWidth: 2,
            pointRadius: 5,
            pointHoverRadius: 8,
            tension: 0.25
        }));

        new Chart(comparisonCanvas, {
            type: "line",
            data: {
                labels: labels,
                datasets: comparisonDatasets
            },
            options: {
                responsive: true,
                maintainAspectRatio: false,
                interaction: {
                    mode: "index",
                    intersect: false
                },
                plugins: {
                    legend: {
                        position: "bottom"
                    },
                    tooltip: {
                        callbacks: {
                            label: function (context) {
                                return context.dataset.label + ": " +
                                    formatPercent(context.parsed.y);
                            }
                        }
                    }
                },
                scales: {
                    y: {
                        beginAtZero: true,
                        ticks: {
                            callback: function (value) {
                                return value + "%";
                            }
                        }
                    },
                    x: {
                        title: {
                            display: true,
                            text: "Tahun"
                        }
                    }
                }
            }
        });
    }
